<?php
/**
 * WordPress Verify Checksums
 * Simpan sebagai: verify-checksums.php
 * Letakkan di root WordPress
 */

define('WP_USE_THEMES', false);
require_once __DIR__ . '/wp-load.php';
require_once ABSPATH . WPINC . '/version.php';

header('Content-Type: text/plain');

global $wp_version, $wp_local_package;

$locale = empty($wp_local_package) ? 'en_US' : $wp_local_package;

$url = "https://api.wordpress.org/core/checksums/1.0/?version={$wp_version}&locale={$locale}";

$json = file_get_contents($url);

if (!$json) {
    exit("ERROR: Tidak bisa mengambil checksum dari WordPress.org\n");
}

$data = json_decode($json, true);

if (empty($data['checksums'])) {
    exit("ERROR: Checksum tidak ditemukan.\n");
}

$checksums = $data['checksums'];

$ok = 0;
$modified = 0;
$missing = 0;

foreach ($checksums as $file => $hash) {

    $path = ABSPATH . $file;

    if (!file_exists($path)) {
        echo "[MISSING] $file\n";
        $missing++;
        continue;
    }

    $md5 = md5_file($path);

    if (strcasecmp($md5, $hash) !== 0) {
        echo "[MODIFIED] $file\n";
        $modified++;
    } else {
        $ok++;
    }
}

echo "\n====================\n";
echo "Version  : $wp_version\n";
echo "Locale   : $locale\n";
echo "OK       : $ok\n";
echo "Modified : $modified\n";
echo "Missing  : $missing\n";
echo "====================\n";

if ($modified == 0 && $missing == 0) {
    echo "Success: WordPress installation verifies against checksums.\n";
} else {
    echo "Warning: Checksum verification failed.\n";
}
